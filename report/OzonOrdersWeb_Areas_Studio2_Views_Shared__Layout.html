 544            const stored = localStorage.getItem('chatMessages');

 551            const filteredMsgs = msgs.filter(msg => {

 558                            const msgDate = new Date(
 559                                parseInt(dateParts[2]), // год
 560                                parseInt(dateParts[1]) - 1, // месяц (0-11)
 561                                parseInt(dateParts[0]) // день
 562                            );
 563                            return msgDate >= thirtyDaysAgo;
 564                        }
 565                    }
 566                    // Если не удалось распарсить дату, сохраняем сообщение
 567                    return true;
 568                } catch {
 569                    // Сохраняем сообщения с некорректной датой
 570                    return true;
 571                }
 572            });
 573
 574            // Сохраняем только если что-то изменилось
 575            if (filteredMsgs.length !== msgs.length) {
 576                localStorage.setItem('chatMessages', JSON.stringify(filteredMsgs));
 577                console.log(`Очищено ${msgs.length - filteredMsgs.length} старых сообщений`);
 578            }
